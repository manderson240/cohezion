## Implementation: SurrealSkillRepository

### create(skill: dict) -> bool
```python
async def create(self, skill: dict[str, Any]) -> bool:
    try:
        skill_id = skill.get("id", f"skill_{int(time.time() * 1000)}")
        await self._client.create("skills", {"id": skill_id, **skill})
        logger.info("Created skill %s", skill_id)
        return True
    except Exception as e:
        logger.error("Failed to create skill: %s", e)
        return False
```

### get(skill_id: str) -> dict | None
```python
async def get(self, skill_id: str) -> dict[str, Any] | None:
    try:
        result = await self._client.query(
            "SELECT * FROM skills WHERE id = $id LIMIT 1",
            {"id": skill_id},
        )
        if result and result[0]:
            records = result[0] if isinstance(result[0], list) else result[0].get("result", [])
            return records[0] if records else None
        return None
    except Exception as e:
        logger.error("Failed to get skill %s: %s", skill_id, e)
        return None
```

### get_all(limit: int = 100) -> list[dict]
```python
async def get_all(self, limit: int = 100) -> list[dict[str, Any]]:
    try:
        result = await self._client.query(
            "SELECT * FROM skills LIMIT $limit",
            {"limit": min(limit, 10000)},
        )
        if result and result[0]:
            records = result[0] if isinstance(result[0], list) else result[0].get("result", [])
            return records
        return []
    except Exception as e:
        logger.error("Failed to get all skills: %s", e)
        return []
```

### Mocking strategy
The `_client` parameter is typed as `Any`, so mock it with AsyncMock:
```python
@pytest.fixture
def mock_client():
    client = AsyncMock()
    client.create = AsyncMock(return_value=[{"id": "skills:test_1"}])
    client.query = AsyncMock(return_value=[[{"id": "test_1", "name": "research"}]])
    return client
```
