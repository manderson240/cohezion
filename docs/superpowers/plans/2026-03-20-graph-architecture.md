```python
#!/usr/bin/env python3
"""
Shared SurrealDB client for vault graph scripts.

Reads connection config from environment variables with root:root fallback
for local development. Production uses SURREALDB_USERNAME/PASSWORD from
the systemd service environment.
"""

import asyncio
import os
from typing import Any

try:
    from surrealdb import Surreal
except ImportError:
    Surreal = None  # type: ignore[assignment]


SURREAL_NS = "cohezion"
SURREAL_DB = "vault"


class GraphQueryError(Exception):
    """Raised when SurrealDB returns an error response."""


class SurrealClient:
    """Thin wrapper around surrealdb.py SDK with env-based config."""

    def __init__(self) -> None:
        self.url = os.environ.get("SURREALDB_URL", "http://localhost:8001")
        self.username = os.environ.get("SURREALDB_USERNAME", "root")
        self.password = os.environ.get("SURREALDB_PASSWORD", "root")
        self.namespace = os.environ.get("SURREALDB_NAMESPACE", SURREAL_NS)
        self.database = os.environ.get("SURREALDB_DATABASE", SURREAL_DB)

    async def _connect(self) -> Any:
        db = Surreal(self.url)
        await db.connect()
        await db.signin({"username": self.username, "password": self.password})
        await db.use(self.namespace, self.database)
        return db

    async def query_async(self, sql: str) -> list:
        """Execute SurrealQL, return result list. Raises GraphQueryError on DB errors."""
        db = await self._connect()
        try:
            data = await db.query(sql)
        finally:
            await db.close()

        if isinstance(data, dict) and "code" in data:
            raise GraphQueryError(data.get("description", str(data)))
        return data

    def query(self, sql: str) -> list:
        """Synchronous wrapper for use in CLI scripts."""
        return asyncio.run(self.query_async(sql))

    async def query_silent_async(self, sql: str) -> list | None:
        """Like query_async but returns None on any failure (for hooks)."""
        try:
            return await self.query_async(sql)
        except Exception:
            return None

    def query_silent(self, sql: str) -> list | None:
        """Synchronous silent wrapper for hooks — never raises."""
        try:
            return asyncio.run(self.query_silent_async(sql))
        except Exception:
            return None


# Singleton for script use
_client: SurrealClient | None = None


def get_client() -> SurrealClient:
    global _client
    if _client is None:
        _client = SurrealClient()
    return _client
```

```python
#!/usr/bin/env python3
"""
GraphReactor — computes dimensional scores for vault neurons and writes
them back to SurrealDB. Also generates metabolism/graph-briefing.md.

Usage:
    python scripts/graph_reactor.py          # compute + write dims + briefing
    python scripts/graph_reactor.py --dims   # dims only, skip briefing
    python scripts/graph_reactor.py --brief  # briefing only, skip dim compute
"""

import datetime
import math
import os
import sys
from pathlib import Path
from typing import Any

import networkx as nx

from graph_client import get_client

VAULT_ROOT = Path(__file__).parent.parent
BRIEFING_PATH = VAULT_ROOT / "metabolism" / "graph-briefing.md"
LAMBDA_DECAY = 0.05  # recency half-life ~14 days


# ── Dimension computation ──────────────────────────────────────────────────────


def compute_bridging_scores(neuron_ids: list[str], edges: list[tuple[str, str]]) -> dict[str, float]:
    """Betweenness centrality normalized to 0-1."""
    G = nx.DiGraph()
    G.add_nodes_from(neuron_ids)
    G.add_edges_from(edges)
    raw = nx.betweenness_centrality(G, normalized=True)
    # Already 0-1 when normalized=True
    return {node: round(score, 4) for node, score in raw.items()}


def compute_completion_score(neuron: dict[str, Any]) -> float:
    """Heuristic completeness 0-1 based on word count, tags, stage."""
    word_count = neuron.get("word_count") or 0
    tag_count = len(neuron.get("tags") or [])
    stage = neuron.get("stage", "embryo")

    stage_score = {"embryo": 0.1, "seedling": 0.3, "growing": 0.6, "mature": 1.0}.get(stage, 0.2)
    word_score = min(word_count / 400.0, 1.0)  # 400 words = "complete"
    tag_score = min(tag_count / 4.0, 1.0)       # 4 tags = well-tagged

    return round(0.5 * word_score + 0.3 * stage_score + 0.2 * tag_score, 4)


def compute_recency_score(modified_date: str | None) -> float:
    """Exponential recency decay: exp(-λ * days). λ=0.05 → half-life ~14 days."""
    if not modified_date:
        return 0.0
    try:
        date = datetime.date.fromisoformat(modified_date[:10])
        days = (datetime.date.today() - date).days
        return round(math.exp(-LAMBDA_DECAY * max(days, 0)), 4)
    except (ValueError, TypeError):
        return 0.0


# ── Main reactor ──────────────────────────────────────────────────────────────


def run_dimensions() -> list[dict[str, Any]]:
    """Fetch graph, compute all dim_* scores, write to SurrealDB. Returns neuron list."""
    client = get_client()

    neurons = client.query(
        "SELECT id, title, word_count, tags, stage, modified, activation, cluster_id, "
        "synapse_out, synapse_in FROM neuron;"
    )[0].get("result", [])

    synapses = client.query(
        "SELECT VALUE [string(in), string(out)] FROM synapse WHERE type = 'explicit';"
    )[0].get("result", [])

    neuron_ids = [str(n["id"]) for n in neurons]
    edges = [(str(s[0]), str(s[1])) for s in synapses if len(s) == 2]

    bridging = compute_bridging_scores(neuron_ids, edges)

    for n in neurons:
        nid = str(n["id"])
        dim_b = bridging.get(nid, 0.0)
        dim_c = compute_completion_score(n)
        dim_r = compute_recency_score(n.get("modified"))
        client.query(
            f"UPDATE {nid} SET "
            f"dim_bridging = {dim_b}, "
            f"dim_completion = {dim_c}, "
            f"dim_recency = {dim_r};"
        )
        n["dim_bridging"] = dim_b
        n["dim_completion"] = dim_c
        n["dim_recency"] = dim_r

    return neurons


def run_briefing(neurons: list[dict[str, Any]] | None = None) -> None:
    """Write metabolism/graph-briefing.md from neuron data."""
    if neurons is None:
        client = get_client()
        neurons = client.query(
            "SELECT id, title, activation, cluster_id, stage, dim_bridging, "
            "dim_completion, dim_recency, synapse_out, synapse_in FROM neuron;"
        )[0].get("result", [])

    n_neurons = len(neurons)
    client = get_client()
    n_synapses = client.query("SELECT count() AS n FROM synapse GROUP ALL;")[0].get("result", [{}])[0].get("n", 0)

    top_bridging = sorted(
        [n for n in neurons if n.get("dim_bridging") is not None],
        key=lambda x: x.get("dim_bridging", 0),
        reverse=True,
    )[:5]

    stubs = sorted(
        [n for n in neurons if (n.get("dim_completion") or 1) < 0.4 and (n.get("dim_bridging") or 0) > 0.05],
        key=lambda x: x.get("dim_bridging", 0),
        reverse=True,
    )[:3]

    recent = sorted(
        [n for n in neurons if n.get("dim_recency") is not None],
        key=lambda x: x.get("dim_recency", 0),
        reverse=True,
    )[:5]

    now = datetime.datetime.now(datetime.timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ")

    lines = [
        f"## Graph State — {now}",
        "",
        f"**Vault:** {n_neurons} neurons · {n_synapses} synapses",
        "",
        "**Hot neurons** (high bridging):",
    ]
    for n in top_bridging:
        lines.append(f"- {n.get('title', '?')} (bridging={n.get('dim_bridging', 0):.3f})")

    lines += ["", "**Completion gaps** (low completion, high bridging — worth fleshing out):"]
    for n in stubs:
        lines.append(f"- {n.get('title', '?')} (completion={n.get('dim_completion', 0):.2f})")

    lines += ["", "**Recent activity**:"]
    for n in recent:
        lines.append(f"- {n.get('title', '?')} (recency={n.get('dim_recency', 0):.3f})")

    BRIEFING_PATH.write_text("\n".join(lines) + "\n")
    print(f"Briefing written to {BRIEFING_PATH}", file=sys.stderr)


def main() -> None:
    args = sys.argv[1:]
    dims_only = "--dims" in args
    brief_only = "--brief" in args

    if brief_only:
        run_briefing()
    elif dims_only:
        run_dimensions()
    else:
        neurons = run_dimensions()
```

```python
#!/usr/bin/env python3
"""
Bootstrap PCA matrix for FLUME affinity pipeline.

Fetches all neuron titles from SurrealDB, embeds them via Ollama nomic-embed-text,
runs PCA to get a 12D projection matrix, saves to scripts/dba/pca_matrix.npy.

Usage:
    python scripts/compute_pca_matrix.py           # full bootstrap
    python scripts/compute_pca_matrix.py --check   # show matrix info
"""

import json
import sys
import urllib.request
from pathlib import Path
from typing import Any

import numpy as np
from sklearn.decomposition import PCA  # type: ignore[import]

from graph_client import get_client

PCA_PATH = Path(__file__).parent / "dba" / "pca_matrix.npy"
OLLAMA_URL = "http://localhost:11434"
EMBED_MODEL = "nomic-embed-text"


def embed_text(text: str) -> list[float]:
    """Get 768D embedding from Ollama nomic-embed-text."""
    payload = json.dumps({"model": EMBED_MODEL, "prompt": text}).encode()
    req = urllib.request.Request(
        f"{OLLAMA_URL}/api/embeddings",
        data=payload,
        headers={"Content-Type": "application/json"},
    )
    with urllib.request.urlopen(req, timeout=30) as resp:
        data = json.loads(resp.read())
    return data["embedding"]


def project_to_12d(vec_768: list[float], pca_matrix: np.ndarray) -> list[float]:
    """Project 768D vector to 12D using pre-computed PCA matrix."""
    v = np.array(vec_768, dtype=np.float32)
    projected = pca_matrix @ v
    return projected.tolist()


def normalize_l2(vec: list[float]) -> list[float]:
    """L2-normalize a vector."""
    v = np.array(vec, dtype=np.float32)
    norm = np.linalg.norm(v)
    if norm < 1e-8:
        return vec
    return (v / norm).tolist()


def fit_pca(embeddings: np.ndarray) -> np.ndarray:
    """Fit PCA on N×768 matrix, return 12×768 projection matrix."""
    pca = PCA(n_components=12, random_state=42)
    pca.fit(embeddings)
    return pca.components_  # shape (12, 768)


def main() -> None:
    if "--check" in sys.argv:
        if PCA_PATH.exists():
            matrix = np.load(PCA_PATH)
            print(f"PCA matrix: {matrix.shape}, dtype={matrix.dtype}")
        else:
            print("No PCA matrix found at", PCA_PATH)
        return

    client = get_client()
    neurons = client.query("SELECT id, title FROM neuron;")[0].get("result", [])
    print(f"Embedding {len(neurons)} neurons...", file=sys.stderr)

    embeddings = []
    for i, n in enumerate(neurons):
        if i % 50 == 0:
            print(f"  {i}/{len(neurons)}...", file=sys.stderr)
        emb = embed_text(n.get("title", ""))
        embeddings.append(emb)

    matrix = np.array(embeddings, dtype=np.float32)
    pca_matrix = fit_pca(matrix)
    np.save(PCA_PATH, pca_matrix)
    print(f"PCA matrix saved: {pca_matrix.shape} → {PCA_PATH}", file=sys.stderr)


if __name__ == "__main__":
    main()
```

```python
"""
Agent-write MCP tools for the vault graph.

Agents may write: latent synapses, dream synapses, affinity vectors,
and metadata annotations. They may NOT overwrite explicit (structural) synapses
or structural neuron fields (title, path, tags, stage).
"""

import logging
from typing import Any

logger = logging.getLogger(__name__)


class GraphWriteError(Exception):
    """Raised when a graph write would violate write boundaries."""


def _validate_not_explicit_synapse(client: Any, from_id: str, to_id: str) -> None:
    """Raise if an explicit synapse already exists between these neurons."""
    sql = (
        f"SELECT type FROM synapse "
        f"WHERE in = {from_id} AND out = {to_id} AND type = 'explicit';"
    )
    results = client.query(sql)
    hits = results[0].get("result", [])
    if hits:
        raise GraphWriteError(
            f"An explicit synapse already exists between {from_id} and {to_id}. "
            "Agents may not overwrite structural synapses."
        )


def build_latent_synapse_sql(from_id: str, to_id: str, reason: str) -> str:
    reason_esc = reason.replace("'", "\\'")
    return (
        f"RELATE {from_id}->{to_id} SET type = 'latent', reason = '{reason_esc}', "
        f"created_at = time::now();"
    )


def build_dream_synapse_sql(from_id: str, to_id: str, resonance: str) -> str:
    resonance_esc = resonance.replace("'", "\\'")
    return (
        f"RELATE {from_id}->{to_id} SET type = 'dream', resonance = '{resonance_esc}', "
        f"created_at = time::now();"
    )


def build_affinity_update_sql(neuron_id: str, vec_12d: list[float]) -> str:
    vec_str = "[" + ", ".join(str(v) for v in vec_12d) + "]"
    return f"UPDATE {neuron_id} SET dim_agent_affinity = {vec_str};"


def build_annotate_sql(
    neuron_id: str,
    last_accessed: str | None = None,
    agent_notes: str | None = None,
    access_count_delta: int = 0,
) -> str:
    sets = []
    if last_accessed:
        sets.append(f"last_accessed = '{last_accessed}'")
    if agent_notes:
        notes_esc = agent_notes.replace("'", "\\'")
        sets.append(f"agent_notes = '{notes_esc}'")
    if access_count_delta:
        sets.append(f"access_count += {access_count_delta}")
    if not sets:
        return ""
    return f"UPDATE {neuron_id} SET {', '.join(sets)};"


def register_graph_write_tools(mcp: Any, get_surrealdb_client: Any) -> None:
    """Register the four agent-write tools with the FastMCP instance."""

    @mcp.tool()
    def graph_write_latent_synapse(from_neuron_id: str, to_neuron_id: str, reason: str) -> str:
        """Create a latent (semantically inferred) synapse between two neurons.

        Args:
            from_neuron_id: Source neuron ID (e.g. 'neuron:cortex_foo_md')
            to_neuron_id: Target neuron ID
            reason: Why this connection exists (stored on the synapse)
        """
        client = get_surrealdb_client()
        _validate_not_explicit_synapse(client, from_neuron_id, to_neuron_id)
        sql = build_latent_synapse_sql(from_neuron_id, to_neuron_id, reason)
        client.query(sql)
        return f"Latent synapse created: {from_neuron_id} -> {to_neuron_id}"

    @mcp.tool()
    def graph_write_dream_synapse(from_neuron_id: str, to_neuron_id: str, resonance: str) -> str:
        """Create a dream synapse (cross-domain resonance) between two neurons.

        Args:
            from_neuron_id: Source neuron ID
            to_neuron_id: Target neuron ID
            resonance: Description of the cross-domain connection
        """
        client = get_surrealdb_client()
        _validate_not_explicit_synapse(client, from_neuron_id, to_neuron_id)
        sql = build_dream_synapse_sql(from_neuron_id, to_neuron_id, resonance)
        client.query(sql)
        return f"Dream synapse created: {from_neuron_id} -> {to_neuron_id}"

    @mcp.tool()
    def graph_write_affinity(neuron_id: str, affinity_vector: list[float]) -> str:
        """Write a 12D FLUME affinity vector to a neuron.

        Args:
            neuron_id: Target neuron ID
            affinity_vector: 12-element float list (L2-normalized)
        """
        if len(affinity_vector) != 12:
            return f"Error: affinity_vector must have exactly 12 elements, got {len(affinity_vector)}"
        client = get_surrealdb_client()
        sql = build_affinity_update_sql(neuron_id, affinity_vector)
        client.query(sql)
        return f"Affinity vector written to {neuron_id}"

    @mcp.tool()
    def graph_annotate_neuron(
        neuron_id: str,
        last_accessed: str = "",
        agent_notes: str = "",
        increment_access_count: bool = False,
    ) -> str:
        """Annotate a neuron with agent metadata (does not touch structural fields).

        Args:
            neuron_id: Target neuron ID
            last_accessed: ISO date string (e.g. '2026-03-20')
            agent_notes: Free-form notes from the agent session
            increment_access_count: Whether to increment access_count by 1
        """
        client = get_surrealdb_client()
        sql = build_annotate_sql(
            neuron_id,
            last_accessed=last_accessed or None,
            agent_notes=agent_notes or None,
            access_count_delta=1 if increment_access_count else 0,
        )
        if not sql:
            return "No fields to update."
        client.query(sql)
        return f"Annotations written to {neuron_id}"
```

```python
"""Async SurrealDB client for vault_graph module."""

import os
from typing import Any

from surrealdb import Surreal


class GraphQueryError(Exception):
    """SurrealDB returned an error response."""


class GraphClient:
    def __init__(self) -> None:
        self.url = os.environ.get("SURREALDB_URL", "http://localhost:8001")
        self.username = os.environ.get("SURREALDB_USERNAME", "root")
        self.password = os.environ.get("SURREALDB_PASSWORD", "root")
        self.namespace = os.environ.get("SURREALDB_NAMESPACE", "cohezion")
        self.database = os.environ.get("SURREALDB_DATABASE", "vault")

    async def _make_connection(self) -> Any:
        db = Surreal(self.url)
        await db.connect()
        await db.signin({"username": self.username, "password": self.password})
        await db.use(self.namespace, self.database)
        return db

    async def query(self, sql: str) -> list:
        """Execute SurrealQL, unwrap first result set, raise on error."""
        db = await self._make_connection()
        try:
            data = await db.query(sql)
        finally:
            await db.close()
        if isinstance(data, dict) and "code" in data:
            raise GraphQueryError(data.get("description", str(data)))
        # Unwrap: [{"result": [...]}] → [...]
        if data and isinstance(data, list) and "result" in data[0]:
            return data[0]["result"]
        return data or []

    async def execute(self, sql: str) -> None:
        """Execute SurrealQL without returning results (for writes)."""
        await self.query(sql)


_client: GraphClient | None = None


def get_graph_client() -> GraphClient:
    global _client
    if _client is None:
        _client = GraphClient()
    return _client
```

```python
"""FLUME PCA affinity pipeline — importable module extracted from compute_pca_matrix.py."""

from pathlib import Path
from typing import Any

import numpy as np

PCA_PATH = Path(__file__).parent.parent.parent.parent.parent / "vaults" / "cohezion-vault" / "scripts" / "dba" / "pca_matrix.npy"
# Override via env var for portability:
import os
_pca_env = os.environ.get("VAULT_PCA_MATRIX_PATH")
if _pca_env:
    PCA_PATH = Path(_pca_env)


def load_pca_matrix() -> np.ndarray | None:
    """Load the pre-computed PCA matrix. Returns None if not found."""
    if PCA_PATH.exists():
        return np.load(str(PCA_PATH)).astype(np.float32)
    return None


def project_to_12d(vec_768: list[float], pca_matrix: np.ndarray) -> list[float]:
    """Project a 768D embedding vector to 12D using the PCA matrix."""
    v = np.array(vec_768, dtype=np.float32)
    return (pca_matrix @ v).tolist()


def normalize_l2(vec: list[float]) -> list[float]:
    """L2-normalize a vector. Returns zeros if input is zero vector."""
    v = np.array(vec, dtype=np.float32)
    norm = float(np.linalg.norm(v))
    if norm < 1e-8:
        return [0.0] * len(vec)
    return (v / norm).tolist()


def embed_and_project(text: str, pca_matrix: np.ndarray, ollama_url: str = "http://localhost:11434") -> list[float] | None:
    """Embed text via Ollama nomic-embed-text, project to 12D. Returns None on failure."""
    import json
    import urllib.request
    try:
        payload = json.dumps({"model": "nomic-embed-text", "prompt": text}).encode()
        req = urllib.request.Request(
            f"{ollama_url}/api/embeddings",
            data=payload,
            headers={"Content-Type": "application/json"},
        )
        with urllib.request.urlopen(req, timeout=10) as resp:
            emb = json.loads(resp.read())["embedding"]
        projected = project_to_12d(emb, pca_matrix)
        return normalize_l2(projected)
    except Exception:
        return None
```

```python
"""GraphReactor — importable module extracted from scripts/graph_reactor.py.

Contains the pure computation logic. The scripts/graph_reactor.py CLI wrapper
delegates to this module.
"""

import datetime
import math
from pathlib import Path
from typing import Any

import networkx as nx

LAMBDA_DECAY = 0.05


def compute_bridging_scores(neuron_ids: list[str], edges: list[tuple[str, str]]) -> dict[str, float]:
    """Betweenness centrality normalized 0-1."""
    G = nx.DiGraph()
    G.add_nodes_from(neuron_ids)
    G.add_edges_from(edges)
    raw = nx.betweenness_centrality(G, normalized=True)
    return {node: round(score, 4) for node, score in raw.items()}


def compute_completion_score(neuron: dict[str, Any]) -> float:
    """Heuristic completeness 0-1."""
    word_count = neuron.get("word_count") or 0
    tag_count = len(neuron.get("tags") or [])
    stage = neuron.get("stage", "embryo")
    stage_score = {"embryo": 0.1, "seedling": 0.3, "growing": 0.6, "mature": 1.0}.get(stage, 0.2)
    word_score = min(word_count / 400.0, 1.0)
    tag_score = min(tag_count / 4.0, 1.0)
    return round(0.5 * word_score + 0.3 * stage_score + 0.2 * tag_score, 4)


def compute_recency_score(modified_date: str | None) -> float:
    """Exponential recency decay exp(-0.05 * days). Half-life ~14 days."""
    if not modified_date:
        return 0.0
    try:
        date = datetime.date.fromisoformat(modified_date[:10])
        days = (datetime.date.today() - date).days
        return round(math.exp(-LAMBDA_DECAY * max(days, 0)), 4)
    except (ValueError, TypeError):
        return 0.0


def build_briefing(neurons: list[dict[str, Any]], n_synapses: int) -> str:
    """Build the metabolism/graph-briefing.md content string."""
    now = datetime.datetime.now(datetime.timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ")
    n_neurons = len(neurons)

    top_bridging = sorted(
        [n for n in neurons if n.get("dim_bridging") is not None],
        key=lambda x: x.get("dim_bridging", 0), reverse=True,
    )[:5]

    stubs = sorted(
        [n for n in neurons if (n.get("dim_completion") or 1) < 0.4 and (n.get("dim_bridging") or 0) > 0.05],
        key=lambda x: x.get("dim_bridging", 0), reverse=True,
    )[:3]

    recent = sorted(
        [n for n in neurons if n.get("dim_recency") is not None],
        key=lambda x: x.get("dim_recency", 0), reverse=True,
    )[:5]

    lines = [
        f"## Graph State — {now}",
        "",
        f"**Vault:** {n_neurons} neurons · {n_synapses} synapses",
        "",
        "**Hot neurons** (high bridging):",
    ]
    for n in top_bridging:
        lines.append(f"- {n.get('title', '?')} (bridging={n.get('dim_bridging', 0):.3f})")

    lines += ["", "**Bridges** (connect otherwise-disconnected clusters):"]
    for n in top_bridging[:3]:
        lines.append(f"- {n.get('title', '?')} → {n.get('cluster_id', '?')}")

    lines += ["", "**Completion gaps** (low completion, high bridging — worth fleshing out):"]
    for n in stubs:
        lines.append(f"- {n.get('title', '?')} (completion={n.get('dim_completion', 0):.2f})")

    lines += ["", "**Recent activity**:"]
    for n in recent:
        lines.append(f"- {n.get('title', '?')} (recency={n.get('dim_recency', 0):.3f})")

    return "\n".join(lines) + "\n"
```
