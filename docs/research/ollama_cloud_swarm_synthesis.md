```python
def kolmogorov_monotonicity_projection(
    abnormal_prob: np.ndarray, finding_prob: np.ndarray, eps: float = 1e-4
) -> tuple[np.ndarray, np.ndarray]:
    """Project probabilities onto Kolmogorov cone: P(Abnormal) >= max_k P(Finding_k)."""
    abnormal_prob = np.clip(abnormal_prob, eps, 1.0 - eps)
    finding_prob = np.clip(finding_prob, eps, 1.0 - eps)
    max_finding = np.max(finding_prob, axis=1)
    abnormal_proj = np.maximum(abnormal_prob, max_finding)
    return abnormal_proj, finding_prob
```
