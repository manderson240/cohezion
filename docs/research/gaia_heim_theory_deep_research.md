    * Selector projection P: R^12 -> R^4 (differentiable analogue of Heim's
      selector calculus).
    * 3D spatial Metron lattice: a discrete scalar field representing cellular
      Planck-scale occupancy/curvature.
    """

    def __init__(
        self,
        lattice_res: int = 32,
        metron_area: float = 2.612e-70,  # ~ (Planck length)^2 order, m^2
        device: str = "cpu",
    ):
        super().__init__()
        self.device = device
        self.metron_area = torch.tensor(metron_area, device=device)
        self.res = lattice_res

        # ------------------------------------------------------------------
        # 1. Continuous metric blocks (the "polymetric" G_AB, A,B = 0..11)
        # ------------------------------------------------------------------
        # 4D external block: signature (+, -, -, -)
        self.g_4d_base = nn.Parameter(
            torch.diag(torch.tensor([1.0, -1.0, -1.0, -1.0], device=device))
        )

        # 8D internal block (small initial values to avoid blow-up)
        self.g_internal = nn.Parameter(torch.eye(8, device=device) * 0.01)

        # Coupling between external and internal space
        self.g_coupling = nn.Parameter(torch.randn(4, 8, device=device) * 1e-5)

        # Selector / projection operator: maps 12D coordinates to 4D
        # Heim's "Selektoren" are treated here as a learnable projection matrix.
        self.selector = nn.Parameter(torch.randn(12, 4, device=device))

        # ------------------------------------------------------------------
        # 2. Discrete Metron lattice (3D spatial grid for a given t-slice)
        # ------------------------------------------------------------------
        # Represents the cellular structure of space. Values > 0 indicate
        # metron occupancy/activation; values near 0 indicate "gaps".
        self.register_buffer(
            "metron_lattice",
            torch.ones(lattice_res, lattice_res, lattice_res, device=device),
        )

        # Strength with which the metron lattice modulates the 4D metric
        self.metron_coupling = nn.Parameter(torch.tensor(1e-3, device=device))

    # ------------------------------------------------------------------
    # Sampling the discrete metron field
    # ------------------------------------------------------------------
    def sample_metron_density(self, x_spatial: torch.Tensor) -> torch.Tensor:
        """
        Trilinearly sample the metron lattice at continuous spatial coordinates.

        Args:
            x_spatial: (..., 3) coordinates normalized to [-1, 1].

        Returns:
            density: (...) scalar field values.
        """
        # Flatten batch dimensions for grid_sample
        batch_shape = x_spatial.shape[:-1]
        x_flat = x_spatial.reshape(1, 1, 1, -1, 3)  # (N, D_out, H_out, W_out, 3)

        # lattice shape: (1, 1, D, H, W)
        lattice = self.metron_lattice.unsqueeze(0).unsqueeze(0)

        sampled = F.grid_sample(
            lattice,
            x_flat,
            mode="bilinear",
            padding_mode="border",
            align_corners=False,
        )
        return sampled.view(batch_shape)

    # ------------------------------------------------------------------
    # Build the 12D Polymetric
    # ------------------------------------------------------------------
    def build_12d_polymetric(self, x: torch.Tensor) -> torch.Tensor:
        """
        Assemble the 12x12 metric tensor G_AB at the 4D spacetime point(s) x.

        Args:
            x: (..., 4) coordinates (t, x, y, z).

        Returns:
            G: (..., 12, 12) symmetric polymetric.
        """
        batch_shape = x.shape[:-1]

        # Expand base metrics to batch
        g4 = self.g_4d_base.expand(*batch_shape, 4, 4).clone()
        g8 = self.g_internal.expand(*batch_shape, 8, 8).clone()
        gc = self.g_coupling.expand(*batch_shape, 4, 8).clone()

        # Block assembly: G = [[g4, gc], [gc^T, g8]]
        top = torch.cat([g4, gc], dim=-1)           # (..., 4, 12)
        bottom = torch.cat([gc.transpose(-2, -1), g8], dim=-1)  # (..., 8, 12)
        G = torch.cat([top, bottom], dim=-2)         # (..., 12, 12)

        # Symmetrize
        G = 0.5 * (G + G.transpose(-2, -1))

        # ------------------------------------------------------------------
        # Metron modulation: discrete lattice imprints on the 4D block
        # ------------------------------------------------------------------
        if x.shape[-1] >= 4:
            # Use spatial components (x, y, z) to sample the metron lattice
            density = self.sample_metron_density(x[..., 1:])  # (...)
            # Create a small, signature-preserving perturbation
            scale = self.metron_coupling * torch.tanh(density)
            # Diagonal Lorentzian signature mask for the 4D block
            sig_mask = torch.diag(
                torch.tensor([1.0, -1.0, -1.0, -1.0], device=x.device)
            ).expand(*batch_shape, 4, 4)
            G[..., :4, :4] = G[..., :4, :4] + scale.unsqueeze(-1).unsqueeze(-1) * sig_mask

        return G

    # ------------------------------------------------------------------
    # 12D -> 4D Projection (Selector Calculus)
    # ------------------------------------------------------------------
    def project_to_4d(
        self,
        G: torch.Tensor,
        method: str = "selector",
    ) -> torch.Tensor:
        """
        Project the 12D polymetric to an effective 4D metric.

        Args:
            G: (..., 12, 12) polymetric.
            method: 'selector' uses the learnable Heim-style selector.
                    'slice' simply extracts the upper-left 4x4 block.

        Returns:
            g_eff: (..., 4, 4) effective 4D metric.
        """
        if method == "selector":
            # Normalize columns to keep projection stable
            P = self.selector / (self.selector.norm(dim=0, keepdim=True) + 1e-8)
            # g_{mu nu} = P^A_mu * G_{AB} * P^B_nu
            # einsum: batch dims ... ; indices A,B (12) ; output a,b (4)
            g_eff = torch.einsum("...AB,Aa,Bb->...ab", G, P, P)
            return g_eff
        elif method == "slice":
            return G[..., :4, :4]
        else:
            raise ValueError(f"Unknown projection method: {method}")

    # ------------------------------------------------------------------
    # Discrete Interval (Metron-quantized line element)
    # ------------------------------------------------------------------
    def compute_metron_interval(
        self,
        x: torch.Tensor,
        dx: torch.Tensor,
    ) -> Tuple[torch.Tensor, torch.Tensor]:
        """
        Compute the spacetime interval ds^2, quantized in units of metron_area.

        Args:
            x: Base point (..., 4).
            dx: Displacement (..., 4).

        Returns:
            ds2_continuous: (..., ) continuous interval from projected 4D metric.
            ds2_discrete: (..., ) interval quantized to nearest metron area.
        """
        G = self.build_12d_polymetric(x)
        g4d = self.project_to_4d(G, method="selector")

        # Standard line element: ds^2 = g_{mu nu} dx^mu dx^nu
        ds2_cont = torch.einsum("...i,...ij,...j->...", dx, g4d, dx)

        # Heim's metron concept implies a discrete cellular structure:
        # intervals are integer multiples of the fundamental metron area.
        ds2_disc = torch.round(ds2_cont / self.metron_area) * self.metron_area

        return ds2_cont, ds2_disc

    # ------------------------------------------------------------------
    # Forward pass
    # ------------------------------------------------------------------
    def forward(
        self,
        x: torch.Tensor,
        dx: Optional[torch.Tensor] = None,
    ) -> dict:
        """
        Args:
            x: (..., 4) spacetime coordinates.
            dx: Optional (..., 4) displacement for interval calculation.

        Returns:
            Dictionary containing 12D metric, 4D projection, and discrete interval.
        """
        G = self.build_12d_polymetric(x)
        g4d = self.project_to_4d(G)

        out = {
            "G_12d": G,
            "g_4d_eff": g4d,
            "metron_area": self.metron_area,
            "signature": torch.tensor([1, -1, -1, -1], device=self.device),
        }

        if dx is not None:
            ds2_c, ds2_d = self.compute_metron_interval(x, dx)
            out["ds2_continuous"] = ds2_c
            out["ds2_discrete"] = ds2_d

        return out
